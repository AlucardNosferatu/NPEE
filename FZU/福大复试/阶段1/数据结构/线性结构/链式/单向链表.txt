// 单向链表节点结构体：存储数据+后继节点指针（核心组成）
template <typename T>  // 模板类型，支持任意数据类型（如int、string等）
struct ListNode {
    T val;                // 成员变量1：节点存储的核心数据
    ListNode<T>* next;    // 成员变量2：后继节点指针，指向当前节点的下一个节点（单向链表核心特征）

    // 构造函数1：默认构造（节点数据默认初始化，后继指针置空）
    ListNode() : val(T()), next(nullptr) {}

    // 构造函数2：带数据参数构造（指定节点数据，后继指针置空）
    ListNode(T value) : val(value), next(nullptr) {}

    // 构造函数3：带数据+后继节点构造（指定数据和下一个节点，快速构建链表节点链）
    ListNode(T value, ListNode<T>* next_node) : val(value), next(next_node) {}
};
