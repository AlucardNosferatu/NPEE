// 边集数组：使用结构体数组存储所有边，适用于需要按边操作的场景（如Kruskal算法、边遍历）
struct Edge {
    int from;        // 边的起点顶点编号
    int to;          // 边的终点顶点编号
    int weight;      // 边的权值，默认为1（无权图）
    
    // 默认构造函数：创建一条空边
    Edge() : from(-1), to(-1), weight(0) {}
    
    // 带参数构造函数：快速初始化边的起点、终点和权值
    // 原理：直接为三个成员变量赋值，用于快速构建边对象
    Edge(int f, int t, int w = 1) : from(f), to(t), weight(w) {}
    
    // 重载小于运算符：用于边按权值排序（升序）
    // 原理：比较两条边的权值，支持在排序算法中使用
    bool operator<(const Edge& other) const {
        return weight < other.weight;
    }
};

// ========================================
// 核心操作封装（边集数组类）
// ========================================
template <typename TVertex>
class EdgeList {
private:
    vector<Edge> edges;          // 存储所有边的数组
    vector<TVertex> vertexData;  // 顶点数据数组，可存储顶点额外信息
    int numVertices;             // 顶点数量
    bool directed;               // 是否为有向图
    int numEdges;                // 当前边的数量

public:
    // 构造函数：初始化指定顶点数的边集，边数组为空
    // 原理：设置顶点数量和方向标志，边数组初始为空
    EdgeList(int n, bool isDirected = false) 
        : vertexData(n), numVertices(n), directed(isDirected), numEdges(0) {}

    // 添加边：向边集数组中添加一条边
    // 原理：1. 创建Edge对象；2. 将其push_back到edges数组中；3. 若是无向图，需要添加反向边（除非特别说明）
    void addEdge(int from, int to, int weight = 1);

    // 删除边：从边集数组中删除指定起点和终点的边（单向）
    // 原理：遍历edges数组，找到匹配的边并删除（使用erase-remove惯用法）
    void removeEdge(int from, int to);

    // 判断边是否存在：检查是否存在从from到to的边
    // 原理：遍历edges数组，查找匹配的边
    bool hasEdge(int from, int to) const;

    // 获取边权值：返回指定边的权值，如果不存在则返回特定值（如-1）
    // 原理：遍历edges数组，找到匹配的边并返回其权值
    int getEdgeWeight(int from, int to) const;

    // 设置边权值：修改已存在边的权值
    // 原理：遍历edges数组，找到匹配的边并更新其权值
    void setEdgeWeight(int from, int to, int newWeight);

    // 获取所有边：返回边集数组的只读引用，便于遍历
    // 原理：返回edges的const引用
    const vector<Edge>& getAllEdges() const { return edges; }

    // 获取与顶点相关的边：返回所有以v为起点的边（有向图）或所有与v相连的边（无向图）
    // 原理：遍历edges数组，根据方向标志收集相关边
    vector<Edge> getEdgesFrom(int v) const;

    // 获取与顶点相关的边：返回所有以v为终点的边（有向图）
    // 原理：遍历edges数组，收集to为v的边
    vector<Edge> getEdgesTo(int v) const;

    // 获取顶点总数
    // 原理：返回numVertices成员变量
    int getNumVertices() const { return numVertices; }

    // 获取边总数
    // 原理：返回numEdges成员变量
    int getNumEdges() const { return numEdges; }

    // 判断是否为有向图
    // 原理：返回directed成员变量
    bool isDirected() const { return directed; }

    // 设置顶点数据：为顶点v设置额外数据
    // 原理：在vertexData数组中设置对应位置的数据
    void setVertexData(int v, const TVertex& data);

    // 获取顶点数据：返回顶点v的数据
    // 原理：从vertexData数组中获取对应位置的数据
    TVertex getVertexData(int v) const;

    // 清空所有边：移除边集数组中的所有边
    // 原理：清空edges数组，重置边数量为0
    void clearEdges();

    // 按权值排序：对边集数组按权值升序排序
    // 原理：调用std::sort，使用Edge类中重载的<运算符
    void sortByWeight();

    // 打印所有边（调试用）
    // 原理：遍历edges数组，打印每条边的信息
    void printEdges() const;
};
