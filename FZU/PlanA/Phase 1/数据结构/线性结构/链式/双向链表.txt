// 双向链表节点结构体：存储数据+前驱节点指针+后继节点指针（核心组成）
template <typename T>  // 模板类型，支持任意数据类型（int、string等）
struct DListNode {
    T val;                // 成员变量1：节点存储的核心数据
    DListNode<T>* prev;   // 成员变量2：前驱节点指针，指向当前节点的上一个节点（双向链表核心特征）
    DListNode<T>* next;   // 成员变量3：后继节点指针，指向当前节点的下一个节点

    // 构造函数1：默认构造（数据默认初始化，前驱+后继指针均置空）
    DListNode() : val(T()), prev(nullptr), next(nullptr) {}

    // 构造函数2：带数据参数构造（指定节点数据，前驱+后继指针置空）
    DListNode(T value) : val(value), prev(nullptr), next(nullptr) {}

    // 构造函数3：带数据+前驱+后继节点构造（快速构建双向链表节点链）
    DListNode(T value, DListNode<T>* prev_node, DListNode<T>* next_node) 
        : val(value), prev(prev_node), next(next_node) {}
};
