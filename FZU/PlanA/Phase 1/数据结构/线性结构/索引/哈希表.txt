// 自定义哈希表（链地址法）：使用数组+链表处理哈希冲突，支持自定义键类型和哈希函数
template <typename Key, typename Value, typename Hash = std::hash<Key>, typename KeyEqual = std::equal_to<Key>>
class CustomHashMap {
private:
    // 哈希表节点定义：链表节点存储键值对
    struct HashNode {
        Key key;                     // 存储键
        Value value;                 // 存储值
        HashNode* next;              // 指向下一个节点的指针（解决哈希冲突）

        // 构造函数：初始化键值对和next指针
        HashNode(const Key& k, const Value& v) : key(k), value(v), next(nullptr) {}
    };

    vector<HashNode*> buckets;       // 桶数组，每个桶是一个链表头指针
    int bucketCount;                 // 桶的数量（数组大小）
    int elementCount;                // 哈希表中元素总数
    Hash hasher;                     // 哈希函数对象，用于计算键的哈希值
    KeyEqual keyEqual;               // 键相等比较函数对象
    double loadFactorThreshold;      // 负载因子阈值，超过时触发扩容
    static const int DEFAULT_BUCKET_COUNT = 16;      // 默认桶数
    static const double DEFAULT_LOAD_FACTOR;         // 默认负载因子阈值（通常0.75）

public:
    // 构造函数：初始化桶数组，设置默认哈希函数和相等比较器
    // 原理：创建指定大小的桶数组，每个桶初始化为nullptr，设置负载因子阈值
    CustomHashMap(int initialCapacity = DEFAULT_BUCKET_COUNT, 
                  double loadFactor = DEFAULT_LOAD_FACTOR);

    // 析构函数：释放所有节点内存
    // 原理：遍历所有桶，删除每个链表的所有节点
    ~CustomHashMap();

    // 插入操作：插入键值对，如果键已存在则更新值
    // 原理：1. 计算键的哈希值并对桶数取模得到桶索引；2. 遍历链表查找键是否存在；3. 存在则更新值，不存在则创建新节点插入链表头部
    void insert(const Key& key, const Value& value);

    // 查找操作：根据键查找对应的值，返回是否找到
    // 原理：1. 计算哈希值得到桶索引；2. 遍历对应链表查找键；3. 找到则通过引用参数返回值并返回true，否则返回false
    bool find(const Key& key, Value& value) const;

    // 删除操作：根据键删除键值对
    // 原理：1. 计算哈希值得到桶索引；2. 遍历链表，找到键匹配的节点并删除（调整链表指针）；3. 更新元素计数
    bool erase(const Key& key);

    // 重载[]运算符：支持map[key]语法，如果键不存在则插入默认值
    // 原理：1. 尝试查找键；2. 若不存在则创建新节点（值为默认构造）；3. 返回值的引用
    Value& operator[](const Key& key);

    // 获取元素数量
    int size() const { return elementCount; }

    // 判断是否为空
    bool empty() const { return elementCount == 0; }

    // 清空哈希表：删除所有元素但保留桶数组
    // 原理：遍历所有桶，删除每个链表的所有节点，重置元素计数
    void clear();

    // 判断键是否存在
    bool contains(const Key& key) const;

    // 获取所有键的集合
    vector<Key> keys() const;

    // 获取所有值的集合
    vector<Value> values() const;

    // 获取所有键值对的集合
    vector<pair<Key, Value>> items() const;

private:
    // 内部函数：计算键的桶索引
    // 原理：使用哈希函数计算哈希值，然后对桶数取模（使用&操作优化，前提桶数是2的幂）
    int getBucketIndex(const Key& key) const;

    // 内部函数：扩容哈希表（重新哈希）
    // 原理：1. 创建新的更大的桶数组（通常翻倍）；2. 遍历所有旧桶中的节点，重新计算哈希并插入新桶；3. 释放旧桶数组内存
    void rehash(int newBucketCount);

    // 内部函数：检查并触发扩容
    // 原理：计算当前负载因子（元素数/桶数），若超过阈值则调用rehash
    void checkAndRehash();
};

// ========================================
// 扩展：支持自定义结构体作为键的哈希表
// ========================================
// 自定义结构体示例
struct Person {
    std::string name;
    int age;
    
    // 必须定义相等运算符
    bool operator==(const Person& other) const {
        return name == other.name && age == other.age;
    }
};

// 自定义哈希函数对象（必须重载operator()）
struct PersonHash {
    std::size_t operator()(const Person& p) const {
        // 组合各个字段的哈希值（参考Boost的hash_combine）
        std::size_t h1 = std::hash<std::string>{}(p.name);
        std::size_t h2 = std::hash<int>{}(p.age);
        return h1 ^ (h2 << 1);  // 简单的组合方式
    }
};

// 使用示例：CustomHashMap<Person, std::string, PersonHash> personMap;

// 扩展：开放寻址法哈希表（线性探测）
// ========================================
template <typename Key, typename Value>
class OpenAddressingHashMap {
private:
    // 桶状态枚举：空、已删除、已占用
    enum BucketStatus { EMPTY, DELETED, OCCUPIED };
    
    struct Bucket {
        Key key;
        Value value;
        BucketStatus status;
        
        Bucket() : status(EMPTY) {}
    };
    
    vector<Bucket> table;            // 哈希表数组
    int elementCount;
    int capacity;
    double loadFactorThreshold;
    
public:
    OpenAddressingHashMap(int initialCapacity = 16, double loadFactor = 0.75);
    
    // 插入操作：使用线性探测解决冲突
    // 原理：1. 计算初始哈希位置；2. 循环查找下一个空桶（EMPTY或DELETED状态）；3. 插入或更新键值对
    void insert(const Key& key, const Value& value);
    
    // 查找操作：线性探测直到找到键或遇到空桶
    bool find(const Key& key, Value& value) const;
    
    // 删除操作：标记为DELETED状态（惰性删除）
    bool erase(const Key& key);
    
    // 其他操作类似...
};

// ========================================
// 扩展：线程安全的哈希表（简单互斥锁保护）
// ========================================
template <typename Key, typename Value>
class ThreadSafeHashMap {
private:
    CustomHashMap<Key, Value> map;
    mutable std::mutex mtx;  // 互斥锁，保护所有操作
    
public:
    void insert(const Key& key, const Value& value) {
        std::lock_guard<std::mutex> lock(mtx);
        map.insert(key, value);
    }
    
    bool find(const Key& key, Value& value) const {
        std::lock_guard<std::mutex> lock(mtx);
        return map.find(key, value);
    }
    
    bool erase(const Key& key) {
        std::lock_guard<std::mutex> lock(mtx);
        return map.erase(key);
    }
    
    // 其他操作类似，都需要加锁...
};
