// 邻接矩阵：使用二维数组（向量）表示图，适用于稠密图，支持快速判断任意两顶点间是否存在边
template <typename T>
class AdjacencyMatrix {
private:
    vector<vector<T>> matrix;  // 二维矩阵，matrix[i][j]表示顶点i到顶点j的边权值
    int numVertices;           // 顶点数量，矩阵大小为numVertices × numVertices
    bool directed;             // 是否为有向图（true-有向图，false-无向图）
    T noEdgeValue;             // 表示无边存在的特殊值（如0、-1或INF）

public:
    // 构造函数：初始化n×n的矩阵，所有元素设为noEdge
    // 原理：创建n×n的二维向量，每个元素初始化为表示无边的值
    AdjacencyMatrix(int n, bool isDirected = false, T noEdge = T()) 
        : matrix(n, vector<T>(n, noEdge)), numVertices(n), directed(isDirected), noEdgeValue(noEdge) {}

    // 添加边：在顶点u和v之间添加边，并设置权值weight
    // 原理：1. 检查索引合法性；2. matrix[u][v] = weight；3. 若是无向图则同时设置matrix[v][u] = weight
    void addEdge(int u, int v, T weight = T(1));

    // 删除边：移除顶点u到v之间的边
    // 原理：1. 检查索引合法性；2. matrix[u][v] = noEdgeValue；3. 若是无向图则同时设置matrix[v][u] = noEdgeValue
    void removeEdge(int u, int v);

    // 判断边是否存在：检查顶点u到v之间是否有边
    // 原理：返回matrix[u][v] != noEdgeValue
    bool hasEdge(int u, int v) const;

    // 获取边权值：返回顶点u到v的边权值，无边则返回noEdgeValue
    // 原理：直接返回matrix[u][v]的值
    T getEdgeWeight(int u, int v) const;

    // 设置边权值：修改已存在边的权值
    // 原理：1. 检查边是否存在；2. 更新matrix[u][v] = newWeight；3. 无向图需同步更新
    void setEdgeWeight(int u, int v, T newWeight);

    // 获取邻接顶点：返回顶点v的所有邻接顶点（对于无向图为所有邻居，有向图为出边邻居）
    // 原理：遍历矩阵第v行（有向图）或第v行和第v列（无向图），收集权值不为noEdgeValue的顶点
    vector<int> getAdjacentVertices(int v) const;

    // 获取顶点入度：返回顶点v的入度（有向图）或度数（无向图）
    // 原理：遍历矩阵第v列（有向图）或第v行（无向图），统计非noEdgeValue的边数
    int getInDegree(int v) const;

    // 获取顶点出度：返回顶点v的出度（有向图）或度数（无向图）
    // 原理：遍历矩阵第v行，统计非noEdgeValue的边数
    int getOutDegree(int v) const;

    // 获取顶点总数
    // 原理：返回numVertices成员变量
    int getNumVertices() const { return numVertices; }

    // 判断是否为有向图
    // 原理：返回directed成员变量
    bool isDirected() const { return directed; }

    // 清空图：移除所有边，但不改变顶点数
    // 原理：将矩阵中所有元素重置为noEdgeValue
    void clearEdges();

    // 打印邻接矩阵（调试用，可选）
    // 原理：按行列打印矩阵，显示边权值或特殊标记
    void printMatrix() const;

    // 获取完整矩阵（只读访问，用于算法如Floyd-Warshall）
    // 原理：返回matrix的const引用
    const vector<vector<T>>& getMatrix() const { return matrix; }
};
