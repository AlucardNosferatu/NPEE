// 简化版堆：使用数组（vector）存储，支持自定义比较规则，实现最小堆/最大堆或复杂排序规则
template <typename T, typename Compare = std::less<T>>
class SimpleHeap {
private:
    vector<T> data;          // 使用vector作为底层存储，动态扩容
    Compare comp;            // 比较函数对象，默认std::less<T>生成最大堆（注意：与priority_queue相反）
    int capacity;            // 当前容量（可选，可省略由vector管理）

public:
    // 构造函数：可指定初始容量和比较器
    // 原理：初始化存储数组和比较器，capacity为可选参数（主要用于防止频繁扩容）
    SimpleHeap(int initialCapacity = 0, Compare cmp = Compare()) 
        : comp(cmp) {
        if (initialCapacity > 0) {
            data.reserve(initialCapacity);
        }
    }

    // 建堆操作：从无序数组构建堆（Floyd算法，O(n)）
    // 原理：从最后一个非叶子节点开始，依次对每个节点执行下沉操作
    void buildHeap(const vector<T>& arr);

    // 插入操作：将元素加入堆，并上浮调整
    // 原理：1. 将新元素添加到数组末尾；2. 从该节点开始上浮（与父节点比较并交换，直到满足堆性质）
    void push(const T& value);

    // 移除堆顶：移除堆顶元素，并调整堆
    // 原理：1. 将堆顶元素与末尾元素交换；2. 删除末尾元素（原堆顶）；3. 从堆顶开始下沉调整
    void pop();

    // 获取堆顶元素：返回堆顶元素（不删除）
    // 原理：返回数组第一个元素（索引0处）
    T top() const;

    // 堆大小：返回堆中元素数量
    int size() const { return data.size(); }

    // 判空操作
    bool empty() const { return data.empty(); }

    // 清空堆
    void clear() { data.clear(); }

    // 获取底层数组（只读，用于调试或遍历）
    const vector<T>& getData() const { return data; }

private:
    // 内部辅助函数：上浮操作（从索引i开始向上调整）
    // 原理：比较当前节点与父节点，若不满足堆性质则交换，并继续向上
    void siftUp(int index);

    // 内部辅助函数：下沉操作（从索引i开始向下调整）
    // 原理：比较当前节点与左右子节点，与满足堆性质的较大（或较小）子节点交换，并继续向下
    void siftDown(int index);

    // 内部辅助函数：获取父节点索引
    int parent(int index) const { return (index - 1) / 2; }

    // 内部辅助函数：获取左子节点索引
    int leftChild(int index) const { return 2 * index + 1; }

    // 内部辅助函数：获取右子节点索引
    int rightChild(int index) const { return 2 * index + 2; }
};

// ========================================
// 扩展：支持自定义复杂排序规则的堆（例如按结构体的某个字段排序）
// ========================================
// 自定义比较器示例：按Person结构体的age字段升序（最小堆）
struct Person {
    string name;
    int age;
};

struct AgeComparator {
    bool operator()(const Person& a, const Person& b) const {
        // 返回true表示a应该在b前面（堆顶）
        return a.age > b.age;  // 升序排列，最小堆（注意与默认比较器方向相反）
    }
};

// 使用示例：SimpleHeap<Person, AgeComparator> ageHeap;

// ========================================
// 扩展：支持更新堆中元素值的堆（索引堆，常用于Dijkstra算法）
// ========================================
template <typename T, typename Compare = std::less<T>>
class IndexedHeap {
private:
    vector<T> values;        // 存储元素值
    vector<int> heap;        // 堆数组，存储索引（指向values）
    vector<int> position;    // position[i]表示索引i在heap中的位置（反向索引）
    Compare comp;            // 比较函数对象

public:
    // 构造函数：初始化大小为n的堆，所有索引不在堆中（用-1表示）
    IndexedHeap(int n, Compare cmp = Compare()) 
        : values(n), position(n, -1), comp(cmp) {}

    // 插入/更新操作：将索引index的值设置为val，并调整堆
    // 原理：1. 如果index不在堆中，则插入（添加到堆末尾并上浮）；2. 如果已在堆中，则更新值并调整位置
    void pushOrUpdate(int index, const T& val);

    // 弹出堆顶：移除堆顶元素，返回其索引和值
    // 原理：1. 交换堆顶和末尾元素；2. 移除末尾索引；3. 堆顶下沉调整；4. 更新position映射
    pair<int, T> pop();

    // 判断堆是否为空
    bool empty() const { return heap.empty(); }

    // 获取堆大小
    int size() const { return heap.size(); }

    // 判断索引是否在堆中
    bool contains(int index) const { 
        return index >= 0 && index < position.size() && position[index] != -1; 
    }

    // 获取索引对应的值（必须在堆中）
    T getValue(int index) const { return values[index]; }

private:
    void siftUp(int pos);
    void siftDown(int pos);
};

// ========================================
// 扩展：支持删除任意元素的堆（惰性删除或标记删除）
// ========================================
template <typename T, typename Compare = std::less<T>>
class RemovableHeap {
private:
    SimpleHeap<T, Compare> heap;      // 主堆
    unordered_map<T, int> countMap;   // 记录每个元素出现的次数（用于惰性删除）

public:
    // 插入操作：插入元素，更新计数
    void push(const T& value);

    // 删除操作：标记删除，并不立即从堆中移除
    // 原理：减少计数，如果计数为0则从map中删除，实际删除延迟到pop时进行
    void remove(const T& value);

    // 弹出堆顶：如果堆顶元素已被标记删除，则跳过并弹出下一个
    // 原理：检查堆顶元素在map中的计数，如果<=0则弹出并继续检查下一个
    void pop();

    // 获取堆顶：跳过已被标记删除的堆顶元素
    T top() const;

    // 清空堆和计数
    void clear() { heap.clear(); countMap.clear(); }

    int size() const { return countMap.size(); }  // 注意：返回的是有效元素数量
    bool empty() const { return size() == 0; }
};

// ========================================
// 扩展：固定容量的堆（用于TopK问题，如求最小的K个数）
// ========================================
template <typename T, typename Compare = std::less<T>>
class FixedSizeHeap {
private:
    SimpleHeap<T, Compare> heap;  // 底层堆
    int maxSize;                  // 堆的最大容量（K值）

public:
    // 构造函数：指定最大容量和比较器
    FixedSizeHeap(int k, Compare cmp = Compare()) : maxSize(k), heap(cmp) {}

    // 插入操作：如果堆未满则直接插入；如果堆已满则比较新元素与堆顶
    // 原理：1. 堆大小<K：直接插入；2. 堆大小=K：比较新元素与堆顶，若满足条件则替换堆顶并调整
    void push(const T& value);

    // 获取堆中所有元素（通常为TopK结果）
    vector<T> getElements() const { return heap.getData(); }

    // 其他操作委托给底层堆
    void pop() { heap.pop(); }
    T top() const { return heap.top(); }
    int size() const { return heap.size(); }
    bool empty() const { return heap.empty(); }
    void clear() { heap.clear(); }
};
