// Trie树（字典树）：多叉树结构，用于高效存储和检索字符串集合，空间换时间，前缀匹配效率高
template <typename T>
class TrieNode {
public:
    // 节点定义：每个节点包含多个子节点指针和结束标志
    vector<TrieNode*> children;  // 子节点指针数组，每个索引对应一个字符（如26个小写字母）
    bool isEnd;                   // 是否为一个单词的结尾节点
    T value;                      // 节点存储的数据（可扩展，如词频统计）

    // 默认构造函数：初始化子节点数组和标志位
    // 原理：创建固定大小的指针数组（如26个），初始化为nullptr，isEnd设为false
    TrieNode(int alphabetSize = 26) : children(alphabetSize, nullptr), isEnd(false) {}

    // 带数据构造函数：快速初始化节点数据（如插入时同时设置值）
    // 原理：在默认构造基础上，额外设置节点存储的数据
    TrieNode(const T& val, int alphabetSize = 26) : children(alphabetSize, nullptr), isEnd(false), value(val) {}
};

// ========================================
// 核心操作封装（Trie类）
// ========================================
template <typename T>
class Trie {
private:
    TrieNode<T>* root;  // 根节点指针，不存储具体字符，作为所有单词的起始点
    int alphabetSize;   // 字母表大小，默认26（小写字母）

public:
    // 构造函数：初始化根节点
    Trie(int size = 26) : root(new TrieNode<T>(size)), alphabetSize(size) {}

    // 插入单词：将一个完整单词插入Trie树
    // 原理：1. 从根节点开始，逐字符处理；2. 若字符对应子节点不存在则创建新节点；3. 单词最后一个字符所在节点标记isEnd=true
    void insert(const string& word, const T& value = T());

    // 搜索单词：判断单词是否存在于Trie树中（必须完全匹配）
    // 原理：1. 从根节点开始，逐字符向下查找；2. 若中途节点不存在返回false；3. 最终节点存在且isEnd=true才返回true
    bool search(const string& word) const;

    // 搜索前缀：判断前缀是否存在于Trie树中（部分匹配）
    // 原理：1. 从根节点开始，逐字符向下查找；2. 若中途节点不存在返回false；3. 所有字符都能找到对应节点则返回true（不要求isEnd=true）
    bool startsWith(const string& prefix) const;

    // 获取单词值：返回单词对应的存储数据（如词频）
    // 原理：类似搜索操作，找到单词结尾节点后返回其value成员
    T getValue(const string& word) const;

    // 删除单词：从Trie树中移除一个单词（惰性删除或完全删除）
    // 原理：1. 查找单词结尾节点；2. 若找到则标记isEnd=false（惰性删除）；3. 可选完全删除：递归删除无用节点直至根节点
    void remove(const string& word);

    // 清空树：释放所有节点内存，重置根节点
    // 原理：后序遍历递归删除所有子节点，最后删除根节点并新建空根节点
    void clear();

    // 获取节点数：统计Trie树中节点总数（包括根节点）
    // 原理：DFS遍历整棵树，计数所有节点
    int nodeCount() const;

    // 获取所有单词：返回Trie树中存储的所有单词集合
    // 原理：DFS遍历，遇到isEnd=true的节点时记录从根到该节点的路径字符串
    vector<string> getAllWords() const;
};
